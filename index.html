let aiConversationId = null; let aiConversations = []; async function loadAIConversations(){ if(!sb || !currentUser){ aiConversations=[]; return; } const {data,error}=await sb .from("ai_conversations") .select("id,title,created_at,updated_at") .eq("user_id",currentUser.id) .order("updated_at",{ascending:false}); if(error){ toast("No se pudo cargar el historial de ALE AI: "+error.message); return; } aiConversations=data||[]; if( aiConversationId && !aiConversations.some(c=>c.id===aiConversationId) ){ aiConversationId=null; } if(!aiConversationId && aiConversations.length){ aiConversationId=aiConversations[0].id; } } async function createAIConversation(title="Nueva conversación"){ if(!sb || !currentUser)return null; const {data,error}=await sb .from("ai_conversations") .insert({ user_id:currentUser.id, title:title||"Nueva conversación" }) .select() .single(); if(error){ toast("No se pudo crear la conversación: "+error.message); return null; } aiConversations.unshift(data); aiConversationId=data.id; return data; } async function loadAIMessages(conversationId){ if(!sb || !currentUser || !conversationId)return []; const {data,error}=await sb .from("ai_messages") .select("id,role,content,created_at") .eq("conversation_id",conversationId) .eq("user_id",currentUser.id) .order("created_at",{ascending:true}); if(error){ toast("No se pudieron cargar los mensajes: "+error.message); return []; } return data||[]; } async function saveAIMessage(conversationId,role,content){ if(!sb || !currentUser || !conversationId)return; const {error}=await sb .from("ai_messages") .insert({ conversation_id:conversationId, user_id:currentUser.id, role, content }); if(error){ toast("No se pudo guardar el mensaje: "+error.message); return; } await sb .from("ai_conversations") .update({ updated_at:new Date().toISOString() }) .eq("id",conversationId) .eq("user_id",currentUser.id); } async function deleteAIConversation(id){ if(!sb || !currentUser || !id)return; const {error}=await sb .from("ai_conversations") .delete() .eq("id",id) .eq("user_id",currentUser.id); if(error){ toast("No se pudo eliminar la conversación: "+error.message); return; } aiConversations=aiConversations.filter(c=>c.id!==id); if(aiConversationId===id){ aiConversationId=aiConversations[0]?.id||null; } render(); if(aiConversationId)await renderAIMessages(); } async function startNewAIConversation(){ const c=await createAIConversation("Nueva conversación"); if(!c)return; render(); await renderAIMessages(); } async function openAIConversation(id){ aiConversationId=id; render(); await renderAIMessages(); } async function renderAIMessages(){ const box=$("#messages"); `; return; } box.innerHTML=messages.map(m=>`
